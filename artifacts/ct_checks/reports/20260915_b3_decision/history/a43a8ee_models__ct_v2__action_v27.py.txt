11: def bounded_residual_xy(observation_box, raw_box, query_delta_t,
12:                         radius_base=.5, radius_per_second=.5, radius_max=2.):
13:     """B2 bounded-always、B3 和监督共用同一个实际动作。"""
14:     observation = observation_box.detach()
15:     raw = raw_box.detach()
16:     dt = query_delta_t.detach().reshape(-1)
17:     finite = torch.isfinite(raw).all(1) & torch.isfinite(observation).all(1)
18:     finite = finite & torch.isfinite(dt) & (dt >= 0)
19:     dt = torch.nan_to_num(dt, nan=0., posinf=0., neginf=0.).clamp_min(0)
20:     radius = (radius_base + radius_per_second * dt).clamp(max=radius_max)
21:     residual = torch.nan_to_num(raw[:, :2] - observation[:, :2],
22:                                 nan=0., posinf=0., neginf=0.)
23:     norm = torch.linalg.norm(residual, dim=1)
24:     scale = (radius / norm.clamp_min(1e-6)).clamp(max=1.)
25:     bounded = residual * scale[:, None]
26:     bounded = torch.where(finite[:, None], bounded, torch.zeros_like(bounded))
27:     return bounded, {"residual": residual, "radius": radius, "scale": scale,
28:                      "norm": norm, "finite": finite}

140:         precision_gain = self.expected_precision_gain_head(hidden).squeeze(1).tanh()
